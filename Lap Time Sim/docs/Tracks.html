 27def read_track(filename, reorder=True):

 30    import sys

 34        print(f"Not a DXF file or a generic I/O error.")

 37        print(f"Invalid or corrupted DXF file.")
 38        sys.exit(2)
 39
 40    #Iterate through dxf and calculate start and end points for each entity
 41    entity_list=[e for e in doc.modelspace() if e.dxftype()!='MTEXT'] #MTEXT entities are the "SOLIDWORKS Educational Purposes only" crap in the bottom left corner of DXFs
 42    
 43    if reorder:
 44        start_points=[]
 45        end_points=[]
 46        line_index=0

 48            if entity_list[i].dxftype()=='ARC':

 53                end_points.append(np.array((center[0]+r*np.cos(entity_list[i].dxfattribs()['end_angle']*np.pi/180),center[1]+r*np.sin(entity_list[i].dxfattribs()['end_angle']*np.pi/180))))
 54

 61                print(entity_list[i].dxftype())

 66        i = line_index

 80                delta_end[j%len(entity_list)] = (end_points[j%len(entity_list)][0] - end_point[0])**2 +(end_points[j%len(entity_list)][1] - end_point[1])**2
 81            
 82            if np.min(delta_start) < 0.01:
 83                correct_order.append(np.argmin(delta_start))
 84            elif np.min(delta_end) < 0.01:
 85                correct_order.append(np.argmin(delta_end))
 86                reverse[np.argmin(delta_end)] = 1
 87            else:
 88                print('some ripe bullshit') #happens if the track isn't a loop, otherwise some extremely ripe bs has occured
 89            i = correct_order[-1]
 90

 95        reordered_reverse= np.take(reverse,correct_order)
 96
 97        print('Processed {} DXF entities'.format(len(reordered_entity_list)))
 98        return (reordered_entity_list,  reordered_reverse)
 99    else:
100        return (entity_list, reverse)

107    Runs a simulation of car around track_tuple, generated by read_track()

111    Parameters:
112    -----------

117    engine : str, optional
118        The engine configuration to be used for the simulation. Default is "R6" - this will be undefined, and used for plotting and storing the Drive_Cycle data.
119    track_name : str, optional
120        The name of the track. Default is "Default", and is used to add info to the 'Drive cycle' plot for sake of data labelling.
121    verbose : bool, optional
122        If True, will show the plots.
123    save_drivecycle : bool, optional
124        If True, will save the drive cycle plot as a .png file, and in a .csv for further analysis.
125
126    Adding the engine, track_name and drive_cycle as optional parameters means process_track() can be used the same as as for the Combustion engine (pre 2024), but with the additional parameters for the EV.
127
128    Returns:
129    --------
130    tuple
131        A tuple containing performance metrics such as lap time, energy consumption, and other relevant data.
132
133    Notes:
134    ------
135    - The function assumes that the car object has methods and attributes required for the simulation.
136    - The track_tuple should be formatted correctly, in a closed loop .dxf from CAD.

586def skidpad(car):
587    '''simulates skidpad event. super simple'''
588    r_turn = 8.5 #m, allowing 125mm to inner cones
589    grip_limit_v = car.cornering_speed(r_turn)
590    t = 2*np.pi*r_turn/grip_limit_v #1 full turn, this assumes left turn and right turn are identical in speed
591    return t

593def acceleration(car):
594    '''simulates acceleration event. super simple'''
595    x = np.linspace(0,75,100)
596    dx = x[1]

598    for i in range(1,len(x)):
599        t[i] = t[i-1] + 2/(car.accel_v(x[i-1])+car.accel_v(x[i]))*dx
600    return t[-1]

track_tuple : tuple A tuple containing track data, typically including coordinates and other track-specific parameters. car : object An instance of the car class containing the car's parameters and powertrain. engine : str, optional The engine configuration to be used for the simulation. Default is "R6" - this will be undefined, and used for plotting and storing the Drive_Cycle data. track_name : str, optional The name of the track. Default is "Default", and is used to add info to the 'Drive cycle' plot for sake of data labelling. verbose : bool, optional If True, will show the plots. save_drivecycle : bool, optional If True, will save the drive cycle plot as a .png file, and in a .csv for further analysis.

Adding the engine, track_name and drive_cycle as optional parameters means process_track() can be used the same as as for the Combustion engine (pre 2024), but with the additional parameters for the EV.

Returns:

tuple A tuple containing performance metrics such as lap time, energy consumption, and other relevant data.

Notes:

  • The function assumes that the car object has methods and attributes required for the simulation.
  • The track_tuple should be formatted correctly, in a closed loop .dxf from CAD.
  • The function will show car velocities around the track in a colored 'speed-heat'-map, as well as a separate plot of the drive cycle if verbose is set to True. """
